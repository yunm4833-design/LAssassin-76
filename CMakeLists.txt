cmake_minimum_required(VERSION 3.22)
project(LAssassin76 VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_PATH "" CACHE PATH "Path to a JUCE 8 checkout (optional)")
option(LASSASSIN_INSTALLED_JUCE "Use a preinstalled JUCE package" OFF)
if(LASSASSIN_INSTALLED_JUCE)
    find_package(JUCE 8.0.6 EXACT CONFIG REQUIRED)
elseif(JUCE_PATH)
    add_subdirectory("${JUCE_PATH}" JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
option(LASSASSIN_BUILD_STANDALONE "Build the standalone application" ON)
set(formats VST3)
if(LASSASSIN_BUILD_STANDALONE)
    list(APPEND formats Standalone)
endif()
if(APPLE)
    list(APPEND formats AU)
endif()
set(lassassin_manifest TRUE)
if(CMAKE_CROSSCOMPILING)
    set(lassassin_manifest FALSE)
endif()
juce_add_plugin(LAssassin76 COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.lassassin76"
    PLUGIN_MANUFACTURER_CODE LAsn PLUGIN_CODE La76 FORMATS ${formats}
    PRODUCT_NAME "LAssassin-76" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics
    VST3_AUTO_MANIFEST ${lassassin_manifest})
juce_add_binary_data(LAssassinFonts NAMESPACE LAssassinFonts HEADER_NAME LAssassinFonts.h
    SOURCES Assets/Fonts/CinzelDecorative-Bold.ttf Assets/Fonts/ZillaSlab-Bold.ttf)
target_sources(LAssassin76 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(LAssassin76 PRIVATE LAssassinFonts)
if(WIN32)
    # Native MSVC embeds the pragma-specified v6 dependency automatically.
    # Cross builds lack mt.exe and must carry an explicit DLL manifest resource.
    if(CMAKE_CROSSCOMPILING)
        configure_file(Source/Windows/manifest.rc.in "${CMAKE_CURRENT_BINARY_DIR}/LAssassin76-manifest.rc" @ONLY)
        target_sources(LAssassin76_VST3 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/LAssassin76-manifest.rc")
    endif()
    if(MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
        # Scanners may lack a Common Controls v6 activation context. Native dialogs
        # must not make COMCTL32 ordinal 345 a mandatory LoadLibrary dependency.
        target_link_options(LAssassin76_VST3 PRIVATE "/DELAYLOAD:COMCTL32.dll")
        target_link_libraries(LAssassin76_VST3 PRIVATE delayimp)
    endif()
endif()

target_compile_definitions(LAssassin76 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(LAssassin76 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
include(CTest)
if(BUILD_TESTING)
    add_executable(dsp_tests tests/dsp_tests.cpp)
    target_include_directories(dsp_tests PRIVATE Source)
    add_test(NAME dsp_models COMMAND dsp_tests)
    add_executable(integration_tests)
    target_sources(integration_tests PRIVATE tests/integration_tests.cpp)
    target_link_libraries(integration_tests PRIVATE LAssassin76)
    get_target_property(juce_module_includes juce::juce_dsp INTERFACE_INCLUDE_DIRECTORIES)
    target_include_directories(integration_tests PRIVATE ${juce_module_includes})
    target_include_directories(integration_tests PRIVATE Source)
    target_compile_definitions(integration_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    add_test(NAME processor_integration COMMAND integration_tests)
    if(NOT WIN32 AND NOT APPLE)
        add_executable(editor_tests tests/editor_tests.cpp)
        target_link_libraries(editor_tests PRIVATE LAssassin76)
        target_include_directories(editor_tests PRIVATE Source ${juce_module_includes})
        target_compile_definitions(editor_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
        add_test(NAME editor_bindings COMMAND editor_tests)
    endif()
endif()

if(BUILD_TESTING AND WIN32)
    set(VST3_SDK "${JUCE_MODULES_DIR}/juce_audio_processors/format_types/VST3_SDK")
    add_subdirectory(tools/WindowsProbe)
    add_test(NAME windows_vst3_load COMMAND LAssassinProbe "$<TARGET_FILE:LAssassin76_VST3>")
endif()
